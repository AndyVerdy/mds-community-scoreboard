**Goal:** One tool, `find`, where filters compose across every data layer — and where the right to
*filter* on a field is separated in code from the right to *show* it.

**Architecture:** A field registry declares every filterable dimension and its disclosure class (🟢 show
/ 🟡 aggregate / 🔴 internal). A pure-logic module turns a request into a filter plan, matches
candidates, builds reasons, and runs the disclosure engine — all unit-tested with no I/O. A thin route
(`/api/olivia/find`) does auth, Supabase reads and assembly. Phase 1 ships the **who** half with
people/count/breakdown output plus the `event_who` routing fix; the route already parses **what**
filters so phase 2 is a tool-schema change, not a rewrite.

**Spec:** `docs/superpowers/specs/2026-08-22-finder-design.md` (read §5 before touching output code —
R10, Andy 2026-08-22: chat membership is a filter signal for everyone and a disclosure only for that
chat's own members).

**Tech Stack:** TypeScript / Next.js App Router (mds-digest-web on Render) · vitest · Supabase
PostgREST (`digest` schema) · n8n (Olivia workflow) · Python for the apply script and the leak gate.

```ts
// The finder's field registry (#108).
//
// Andy, 2026-08-22: "we process everything, but answers can be gated — revenue is not disclosed, but
// that doesn't mean we can't use it in filters." So every dimension carries a CLASS, and the class
// decides three different rights: may it be filtered on, may it be grouped by, may it be printed
// next to a person's name. OLIVIA_SHAREABLE_FIELDS.md is the binding rulebook; this file is that
// rulebook in code, and the disclosure engine in finder.ts is the only thing allowed to read it.
//
//   show      🟢 filter, group, print beside a name        (the member-card set)
//   aggregate 🟡 filter and group, NEVER beside a name      (group-only fields)
//   internal  🔴 never filter, never return                 (exact revenue, contacts, payments, ids)

export type FieldClass = "show" | "aggregate" | "internal";

export type FieldDef = {
  cls: FieldClass;
  /** which data layer it comes from — documentation for the next reader */
  layer: "people" | "event" | "content" | "video" | "partner" | "forms";
  filterable: boolean;
  groupable: boolean;
  /** who-group or what-group */
  group: "who" | "what";
};

const f = (
  cls: FieldClass, layer: FieldDef["layer"], group: FieldDef["group"],
  opts?: { filterable?: boolean; groupable?: boolean },
): FieldDef => ({
  cls, layer, group,
  filterable: opts?.filterable ?? cls !== "internal",
  groupable: opts?.groupable ?? cls !== "internal",
});

export const FIELDS: Record<string, FieldDef> = {
  // ---- who: 🟢 the member-card set -----------------------------------------
  chat: f("show", "people", "who"),
  // `segment` is the FILTER (reseller, supplements, agency…); `business_model` stays a GROUP
  // dimension over the raw labels so breakdowns keep working, but it is not a filter key.
  segment: f("show", "people", "who", { groupable: false }),
  business_model: f("show", "people", "who", { filterable: false }),
  event: f("show", "event", "who", { groupable: false }),
  city: f("show", "people", "who"),
  state: f("show", "people", "who"),
  country: f("show", "people", "who"),
  band: f("show", "people", "who"),          // revenue TIER only — the figure is internal
  chapter: f("show", "people", "who"),
  niche: f("show", "people", "who"),
  category: f("show", "people", "who"),
  expertise: f("show", "people", "who", { groupable: false }),
  speaker_of: f("show", "video", "who", { groupable: false }),

  // ---- who: 🟡 group-only ---------------------------------------------------
  sku_min: f("aggregate", "people", "who", { groupable: false }),
  brands_min: f("aggregate", "people", "who", { groupable: false }),
  years_selling_min: f("aggregate", "people", "who", { groupable: false }),
  age_band: f("aggregate", "people", "who"),
  form_answer: f("aggregate", "forms", "who", { groupable: false }),

  // ---- who: 🔴 never --------------------------------------------------------
  revenue_exact: f("internal", "people", "who"),
  email: f("internal", "people", "who"),
  phone: f("internal", "people", "who"),
  address: f("internal", "people", "who"),
  stripe: f("internal", "people", "who"),
  persona: f("internal", "people", "who"),
  member_id: f("internal", "people", "who"),
  removal_reason: f("internal", "people", "who"),

  // ---- what: content, video, event, partner (parsed now, served in phase 2) -
  terms: f("show", "content", "what", { groupable: false }),
  sources: f("show", "content", "what"),
  kinds: f("show", "content", "what"),
  since: f("show", "content", "what", { groupable: false }),
  until: f("show", "content", "what", { groupable: false }),
  call_type: f("show", "video", "what"),
  video_category: f("show", "video", "what"),
  speaker: f("show", "video", "what"),
  year: f("show", "content", "what"),
  event_type: f("show", "event", "what"),
  partner_category: f("show", "partner", "what"),
  featured: f("show", "partner", "what", { groupable: false }),
};

export const WHO_KEYS = Object.keys(FIELDS).filter((k) => FIELDS[k].group === "who");
export const WHAT_KEYS = Object.keys(FIELDS).filter((k) => FIELDS[k].group === "what");
export const GROUP_KEYS = Object.keys(FIELDS).filter((k) => FIELDS[k].groupable && FIELDS[k].cls !== "internal");

export const classOf = (name: string): FieldClass | null => FIELDS[name]?.cls ?? null;
export const isFilterable = (name: string): boolean => !!FIELDS[name]?.filterable;
export const isGroupable = (name: string): boolean => !!FIELDS[name]?.groupable;
```

```ts
  /** true when any 🟡 field is in play — the disclosure engine reads this */
  usesAggregate: boolean;
};

const lower = (s: string) => s.trim().toLowerCase();
const list = (xs?: string[] | null) => (xs || []).map((x) => String(x).trim()).filter(Boolean);
const num = (v: unknown) => (v === null || v === undefined || v === "" ? null : Number(v));

export function normaliseChats(names: string[] | undefined, known: string[]): {
  chats: string[]; unmatched: string[];
} {
  const chats: string[] = []; const unmatched: string[] = [];
  for (const raw of list(names)) {
    const want = lower(raw).replace(/\bchat\b/g, "").replace(/^mds\s+/, "").trim();
    const hit = known.find((k) => lower(k).replace(/^mds\s+/, "") === want);
    if (hit) { if (!chats.includes(hit)) chats.push(hit); } else { unmatched.push(raw); }
  }
  return { chats, unmatched };
}

export function expandConcepts(who: WhoFilters, knownChats: string[]): FilterPlan {
  const unmatched: string[] = []; const concepts: Concept[] = [];
  for (const raw of list(who.segment)) {
    const key = CONCEPT_SYNONYMS[lower(raw)];
    const found = (key && CONCEPTS.find((c) => c.label === key)) ||
      CONCEPTS.find((c) => c.models.some((m) => lower(m) === lower(raw)));
    if (found) { if (!concepts.includes(found)) concepts.push(found); } else { unmatched.push(raw); }
  }
  const chat = normaliseChats(who.chat, knownChats);
  const skuMin = num(who.sku_min), brandsMin = num(who.brands_min), yearsMin = num(who.years_selling_min);
  const ageBand = list(who.age_band as string[] | undefined);
  return {
    concepts, chats: chat.chats,
    city: list(who.city), state: list(who.state), country: list(who.country),
    band: list(who.band), chapter: list(who.chapter),
    niche: list(who.niche), category: list(who.category),
    expertise: who.expertise ? String(who.expertise).trim() : null,
    event: who.event ? String(who.event).trim() : null,
    speaker_of: who.speaker_of ? String(who.speaker_of).trim() : null,
    skuMin, brandsMin, yearsMin, ageBand,
    unmatched: [...unmatched, ...chat.unmatched],
    usesAggregate: skuMin !== null || brandsMin !== null || yearsMin !== null ||
                   ageBand.length > 0 || !!who.form_answer ||
                   concepts.some((c) => c.cls === "aggregate"),
  };
}

const anyEq = (want: string[], have: (string | null)[]) =>
  want.find((w) => have.some((h) => h && lower(h) === lower(w))) || null;

/**
 * Returns the person with their reasons, or null when any filter kind fails.
 * AND across kinds, OR inside a kind — and a concept is satisfied by ANY of its signals (the union
 * Andy asked for). R3: 🟡 filters gate the match but never become a reason. R10: chat membership
 * is a signal for every asker, but the chat is NAMED only to a member of that chat; for anyone
 * else the reason is the concept word — and only when no other signal already explains the match.
 */
export function matchCandidate(
  c: Candidate, plan: FilterPlan, askerId: string, askerChats: string[],
): FoundPerson | null {
  if (c.at_member_id === askerId) return null;
  const reasons: string[] = [];
  const askerIn = (chat: string) => askerChats.some((x) => lower(x) === lower(chat));

  for (const concept of plan.concepts) {
    const chatHit = concept.chats.find((ch) => c.chats.some((x) => lower(x) === lower(ch))) || null;
    const modelHit = concept.models.find((m) => c.business_model.some((x) => lower(x) === lower(m))) || null;
    const attrHit = concept.attr ? concept.attr(c) : false;
    if (!chatHit && !modelHit && !attrHit) return null;
    const why: string[] = [];
    if (modelHit && concept.modelReason) why.push(concept.modelReason);
    if (attrHit && concept.attrReason) why.push(concept.attrReason);
    if (chatHit) {
      if (askerIn(chatHit)) why.unshift(`in ${chatHit}`);
      else if (!why.length) why.push(concept.label);
    }
    reasons.push(...why);
  }
  for (const ch of plan.chats) {
    const hit = c.chats.find((x) => lower(x) === lower(ch));
    if (!hit) return null;
    // named only to a member of that chat; decideDisclosure withholds names from anyone else
    if (askerIn(hit) && !reasons.includes(`in ${hit}`)) reasons.push(`in ${hit}`);
  }
  if (plan.city.length) { if (!anyEq(plan.city, [c.city])) return null; reasons.push(c.city as string); }
  if (plan.state.length) { if (!anyEq(plan.state, [c.state])) return null; reasons.push(c.state as string); }
  if (plan.country.length) { if (!anyEq(plan.country, [c.country])) return null; reasons.push(c.country as string); }
  if (plan.band.length) { if (!anyEq(plan.band, [c.rev_band])) return null; reasons.push(c.rev_band as string); }
  if (plan.chapter.length) { const h = anyEq(plan.chapter, c.chapter); if (!h) return null; reasons.push(h); }
  if (plan.niche.length) { if (!anyEq(plan.niche, [c.main_niche])) return null; reasons.push(c.main_niche as string); }
  if (plan.category.length) { const h = anyEq(plan.category, c.categories); if (!h) return null; reasons.push(h); }
  if (plan.expertise) {
    const hay = `${c.expertise || ""} ${c.main_niche || ""} ${c.categories.join(" ")}`.toLowerCase();
    if (!hay.includes(lower(plan.expertise))) return null;
    reasons.push(plan.expertise);
  }

  // 🟡 aggregate gates — they decide membership, they never speak (R3).
  if (plan.skuMin !== null && Number(c.sku_count || 0) < plan.skuMin) return null;
  if (plan.brandsMin !== null && Number(c.brands_count || 0) < plan.brandsMin) return null;
  if (plan.yearsMin !== null) {
    const years = c.started_year ? new Date().getFullYear() - c.started_year : -1;
    if (years < plan.yearsMin) return null;
  }
  if (plan.ageBand.length && !anyEq(plan.ageBand, [c.age_band])) return null;

  return {
    at_member_id: c.at_member_id, name: c.full_name || "", reasons: [...new Set(reasons)],
    city: c.city, niche: c.main_niche || c.categories[0] || null,
    engagement_score: c.engagement_score,
  };
}
```

```ts
// ---------------------------------------------------------------------------
// The disclosure engine. Spec §5. Filtering and showing are different rights:
// a 🟡 field may decide WHO is in the set and still never be shown, and naming
// the people a 🟡 filter selected would disclose it anyway — so it doesn't.
// ---------------------------------------------------------------------------

/** #96: a display cap, never a processing cap — `total` always counts everyone. */
export const NAME_CAP = 10;
/** R4: below this, a bucket re-identifies the person the count was meant to protect. */
export const SMALL_BUCKET = 3;

/**
 * #106 (Andy, during #97's prod E2E: "I don't want people to see me as an attendee"). Staff, removed
 * and unknown-status records never reach a member-facing list — and are left out of the count too: a
 * total whose members cannot be shown is a number that lies.
 */
export function isMemberFacing(status: string | null): boolean {
  if (!status) return false;
  const s = status.trim().toLowerCase();
  if (s.startsWith("removed") || s === "staff") return false;
  return s.startsWith("current member") || s === "new member";
}

export function decideDisclosure(x: {
  plan: FilterPlan; askerAttends: boolean; wantNames: boolean; askerChats: string[];
}): { mayName: boolean; reason: "aggregate_filter" | "not_in_chat" | "not_registered" | null } {
  if (!x.wantNames) return { mayName: false, reason: null };
  if (x.plan.usesAggregate) return { mayName: false, reason: "aggregate_filter" };   // R2
  // R10: listing the members of a chat IS saying who is there — only its members may see that.
  const outside = x.plan.chats.find((ch) => !x.askerChats.some((a) => a.toLowerCase() === ch.toLowerCase()));
  if (outside) return { mayName: false, reason: "not_in_chat" };
  if (x.plan.event && !x.askerAttends) return { mayName: false, reason: "not_registered" }; // R5
  return { mayName: true, reason: null };
}

/** Engagement first, then name. Never surfaced — ordering only, and no equalizer (R9). */
export function rank(people: FoundPerson[]): FoundPerson[] {
  return [...people].sort((a, b) => b.engagement_score - a.engagement_score || a.name.localeCompare(b.name));
}

export function applyCap(people: FoundPerson[], limit: number): {
  people: FoundPerson[]; total: number; shown: number; capped: boolean;
} {
  const max = Math.max(1, Math.min(NAME_CAP, Math.floor(limit) || NAME_CAP));
  const ranked = rank(people);
  const shown = ranked.slice(0, max);
  return { people: shown, total: ranked.length, shown: shown.length, capped: ranked.length > shown.length };
}

export type GroupDimension =
  | "country" | "state" | "city" | "band" | "niche" | "business_model" | "chat" | "chapter";
export type Bucket = { value: string; count: number | null; note?: string };

export function groupBy(rows: Candidate[], dimension: GroupDimension, suppressSmall: boolean): Bucket[] {
  const pick = (c: Candidate): string[] => {
    switch (dimension) {
      case "country": return c.country ? [c.country] : [];
      case "state": return c.state ? [c.state] : [];
      case "city": return c.city ? [c.city] : [];
      case "band": return c.rev_band ? [c.rev_band] : [];
      case "niche": return c.main_niche ? [c.main_niche] : c.categories.slice(0, 1);
      case "business_model": return c.business_model;
      case "chat": return c.chats;
      case "chapter": return c.chapter;
    }
  };
  const counts = new Map<string, number>();
  for (const row of rows) for (const v of pick(row)) counts.set(v, (counts.get(v) || 0) + 1);
  return [...counts.entries()]
    .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))
    .map(([value, count]) =>
      suppressSmall && count < SMALL_BUCKET
        ? { value, count: null, note: `fewer than ${SMALL_BUCKET}` }
        : { value, count });
}
```

```ts
export const RETURN_KINDS = ["people", "content", "videos", "events", "partners", "count", "breakdown"] as const;
export type ReturnKind = (typeof RETURN_KINDS)[number];

export type ParsedRequest =
  | { ok: true; phone: string; who: WhoFilters; what: WhatFilters; ret: ReturnKind;
      group_by: GroupDimension | null; limit: number }
  | { ok: false; error: string };

const asList = (v: unknown): string[] =>
  (Array.isArray(v) ? v : v == null || v === "" ? [] : [v]).map((x) => String(x).trim()).filter(Boolean);

/**
 * The allowlist is CLOSED and class-aware: an unknown key is a 400, a 🔴 key is a 400, and a key
 * filed under the wrong group is a 400 that says where it belongs. A tool schema that drifts must
 * never be able to widen what Millie can read.
 */
export function parseRequest(body: unknown): ParsedRequest {
  const b = (body || {}) as Record<string, unknown>;
  const phone = String(b.phone || "").replace(/\D/g, "");
  if (!phone) return { ok: false, error: "phone required" };

  const groups: Array<["who" | "what", Record<string, unknown>]> = [
    ["who", (b.who || {}) as Record<string, unknown>],
    ["what", (b.what || {}) as Record<string, unknown>],
  ];
  for (const [group, raw] of groups) {
    for (const key of Object.keys(raw)) {
      const def = FIELDS[key];
      if (!def) return { ok: false, error: `unknown filter: ${key}` };
      if (def.cls === "internal") return { ok: false, error: `filter not allowed: ${key}` };
      if (!def.filterable) {
        return { ok: false, error: `${key} is a group_by dimension, not a filter — use segment` };
      }
      if (def.group !== group) return { ok: false, error: `${key} belongs in ${def.group}, not ${group}` };
    }
  }
  const rawWho = groups[0][1]; const rawWhat = groups[1][1];

  const who: WhoFilters = {
    chat: asList(rawWho.chat), segment: asList(rawWho.segment),
    city: asList(rawWho.city), state: asList(rawWho.state), country: asList(rawWho.country),
    band: asList(rawWho.band), chapter: asList(rawWho.chapter),
    niche: asList(rawWho.niche), category: asList(rawWho.category),
    event: rawWho.event ? String(rawWho.event).trim() : undefined,
    expertise: rawWho.expertise ? String(rawWho.expertise).trim() : undefined,
    speaker_of: rawWho.speaker_of ? String(rawWho.speaker_of).trim() : null,
    sku_min: rawWho.sku_min == null ? null : Number(rawWho.sku_min),
    brands_min: rawWho.brands_min == null ? null : Number(rawWho.brands_min),
    years_selling_min: rawWho.years_selling_min == null ? null : Number(rawWho.years_selling_min),
    age_band: asList(rawWho.age_band),
    form_answer: rawWho.form_answer ? String(rawWho.form_answer).trim() : null,
  };
  const what: WhatFilters = {
    terms: asList(rawWhat.terms), sources: asList(rawWhat.sources), kinds: asList(rawWhat.kinds),
    chat: asList(rawWhat.chat),
    since: rawWhat.since ? String(rawWhat.since).trim() : null,
    until: rawWhat.until ? String(rawWhat.until).trim() : null,
    call_type: rawWhat.call_type ? String(rawWhat.call_type).trim() : null,
    video_category: asList(rawWhat.video_category), speaker: asList(rawWhat.speaker),
    year: rawWhat.year == null ? null : Number(rawWhat.year),
    event_type: rawWhat.event_type ? String(rawWhat.event_type).trim() : null,
    partner_category: asList(rawWhat.partner_category),
    featured: rawWhat.featured == null ? null : Boolean(rawWhat.featured),
  };

  const filled = (o: Record<string, unknown>) =>
    Object.values(o).some((v) => (Array.isArray(v) ? v.length > 0 : v !== null && v !== undefined && v !== ""));
  if (!filled(who as unknown as Record<string, unknown>) &&
      !filled(what as unknown as Record<string, unknown>)) {
    return { ok: false, error: "at least one filter is required" };
  }

  // The tool schema exposes this as `ret` (the seed is JavaScript, `return` is a keyword);
  // the route accepts either spelling.
  const ret = String(b.return || b.ret || "people") as ReturnKind;
  if (!(RETURN_KINDS as readonly string[]).includes(ret)) return { ok: false, error: `unknown return: ${ret}` };
  const groupRaw = b.group_by ? String(b.group_by) : null;
  if (groupRaw && !GROUP_KEYS.includes(groupRaw)) return { ok: false, error: `unknown group_by: ${groupRaw}` };
  if (ret === "breakdown" && !groupRaw) return { ok: false, error: "breakdown needs group_by" };

  const limit = Number(b.limit) > 0 ? Number(b.limit) : NAME_CAP;
  return { ok: true, phone, who, what, ret, group_by: (groupRaw as GroupDimension) || null, limit };
}
```

```ts
      note: "unknown asker" });

    : null;

      : [],
    items: null,
    breakdown,
    note: buildNote({ disclosure, ret, total: capped.total, shown: capped.shown,
                      unmatched: plan.unmatched, eventName: event?.name || null, what }),
  });
}

function echo(plan: FilterPlan, what: Record<string, unknown>, usesAggregate: boolean) {
  return {
    who: {
      concepts: plan.concepts.map((c) => c.label), chat: plan.chats, city: plan.city,
      state: plan.state, country: plan.country, band: plan.band, chapter: plan.chapter,
      niche: plan.niche, category: plan.category, expertise: plan.expertise, event: plan.event,
    },
    what,
    ...(plan.unmatched.length ? { unmatched: plan.unmatched } : {}),
    disclosure: usesAggregate ? "aggregate" : "green",
  };
}

function buildNote(x: {
  disclosure: { mayName: boolean; reason: string | null }; ret: string;
  total: number; shown: number; unmatched: string[]; eventName: string | null;
  what: Record<string, unknown>;
}): string {
  const parts: string[] = [];
  if (x.unmatched.length) {
    parts.push(`could not resolve ${x.unmatched.join(", ")} — say which part of the ask you could not honour`);
  }
  const whatUsed = Object.values(x.what).some((v) => Array.isArray(v) ? v.length : v !== null && v !== "");
  if (whatUsed) {
    parts.push("this lane cannot filter CONTENT yet — the content filters were ignored; " +
      "use content_search or video_search for what-was-said questions and say plainly that you did");
  }
  if (x.disclosure.reason === "aggregate_filter") {
    parts.push("this filter uses a group-only attribute, so answer with the number and never name " +
      "the people — do not explain internals, just give the count");
  } else if (x.disclosure.reason === "not_in_chat") {
    parts.push("this person is not a member of that chat, so give the count only — never the names and " +
      "never what is discussed there — and do not explain internals");
  } else if (x.disclosure.reason === "not_registered" && x.eventName) {
    parts.push(`this person is not registered for ${x.eventName}, so answer with the count only, ` +
      "never the names, and do not explain internals");
  } else if (x.disclosure.mayName && x.total > x.shown) {
    parts.push(`${x.total} people match; ${x.shown} are listed — the count is the truth, never say ` +
      "the listed names are all of them, and offer the rest");
  }
  parts.push("each person carries the reason they matched — say it in plain words (chat membership is " +
    "behaviour; a business model is what they told MDS on a form). Never quote scores or ranks.");
  return parts.join(". ");
}
```
