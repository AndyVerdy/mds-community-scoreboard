**Goal:** One tool, `find`, where filters compose as a boolean tree across every data layer — and
where the right to *filter* on a field is separated in code from the right to *show* it.

system must be flexible, like filters with groups and conditions, or IFTTT; we don't know how they'll
ask." So the request is a **tree**: `all` (AND) · `any` (OR) · `not`, nesting freely; leaves are
`{ field: value }`; a list value means any-of. No per-field AND/OR rule to remember. Task 1 (the
registry, commit f3aa2ab) stands unchanged; Task 2's first cut (b5dfd93, flat filters) is superseded
by this plan's Task 2 — keep the concept table and synonyms from it.

**Architecture:** A field registry declares every filterable dimension and its disclosure class (🟢
show / 🟡 aggregate / 🔴 internal). A pure-logic module normalises the tree, evaluates it per
candidate returning the *proof* (which leaves held) as reasons, and runs the disclosure engine — all
unit-tested with no I/O. A thin route (`/api/olivia/find`) does auth, loads the (small) member
population and any event rosters, and assembles the answer. Phase 1 serves people / count / breakdown
plus the `event_who` routing fix; what-group leaves are parsed and refused with an honest message
until phase 2.

export function normaliseChats(names: string[] | undefined, known: string[]): {
  chats: string[]; unmatched: string[];
} {
  const chats: string[] = []; const unmatched: string[] = [];
  for (const raw of asList(names)) {
    const want = lower(raw).replace(/\bchat\b/g, "").replace(/^mds\s+/, "").trim();
    const hit = known.find((k) => lower(k).replace(/^mds\s+/, "") === want);
    if (hit) { if (!chats.includes(hit)) chats.push(hit); } else { unmatched.push(raw); }
  }
  return { chats, unmatched };
}

function resolveConcept(raw: string): Concept | null {
  const key = CONCEPT_SYNONYMS[lower(raw)];
  return (key && CONCEPTS.find((c) => c.label === key)) ||

/** The raw request tree is assumed allowlist-valid (parseRequest, Task 4, checks that first). */
export function normaliseTree(raw: unknown, knownChats: string[]): {
  tree: Tree; unmatched: string[]; flags: TreeFlags;
} {
  const unmatched: string[] = [];
  const flags: TreeFlags = { usesAggregate: false, chatLeaves: [], eventNames: [], leafCount: 0, depth: 0 };

  const walk = (node: unknown, depth: number): Tree => {
    flags.depth = Math.max(flags.depth, depth);
    const o = (node || {}) as Record<string, unknown>;
    if (Array.isArray(o.all)) return { all: o.all.map((c) => walk(c, depth + 1)) };
    if (Array.isArray(o.any)) return { any: o.any.map((c) => walk(c, depth + 1)) };
    if (o.not !== undefined) return { not: walk(o.not, depth + 1) };
    const field = Object.keys(o)[0];
    const value = o[field];
    flags.leafCount += 1;
    if (classOf(field) === "aggregate") flags.usesAggregate = true;
    if (NUMERIC_FIELDS.has(field)) return { field, values: [], n: Number(value) };
    if (field === "segment") {
      const concepts: Concept[] = [];
      for (const label of asList(value)) {
        const c = resolveConcept(label);
        if (c) { if (!concepts.includes(c)) concepts.push(c); } else { unmatched.push(label); }
      }
      if (concepts.some((c) => c.cls === "aggregate")) flags.usesAggregate = true;
      return { field, values: concepts.map((c) => c.label), concepts };
    }
    if (field === "chat") {
      const r = normaliseChats(asList(value), knownChats);
      unmatched.push(...r.unmatched);
      flags.chatLeaves.push(...r.chats.filter((c) => !flags.chatLeaves.includes(c)));
      return { field, values: r.chats };
    }
    if (field === "event") {
      const values = asList(value);
      flags.eventNames.push(...values.filter((v) => !flags.eventNames.includes(v)));
      return { field, values };
    }
    return { field, values: asList(value) };
  };

  return { tree: walk(raw, 1), unmatched, flags };
}

export function mapLeaves(tree: Tree, fn: (leaf: Leaf) => Leaf): Tree {
  if ("all" in tree) return { all: tree.all.map((t) => mapLeaves(t, fn)) };
  if ("any" in tree) return { any: tree.any.map((t) => mapLeaves(t, fn)) };
  if ("not" in tree) return { not: mapLeaves(tree.not, fn) };
  return fn(tree);
}

/** The normalised tree back in request shape — what the model re-sends to narrow. */
export function toEcho(tree: Tree): unknown {
  if ("all" in tree) return { all: tree.all.map(toEcho) };
  if ("any" in tree) return { any: tree.any.map(toEcho) };
  if ("not" in tree) return { not: toEcho(tree.not) };
  if (tree.n !== undefined && tree.n !== null) return { [tree.field]: tree.n };
  return { [tree.field]: tree.values.length === 1 ? tree.values[0] : tree.values };
}

// ---------------------------------------------------------------------------
// Evaluation — returns the proof
// ---------------------------------------------------------------------------

export type EvalCtx = {
  askerId: string;
  askerChats: string[];
  /** resolved event name -> member ids on its registration roster */
  eventMembers: Map<string, Set<string>>;
};
export type Verdict = { ok: boolean; reasons: string[] };

const NO: Verdict = { ok: false, reasons: [] };
const eq = (a: string | null | undefined, b: string) => !!a && lower(a) === lower(b);
/** the candidate's own matching value, so the reason carries its casing, not the query's */
const pick = (want: string[], have: (string | null)[]): string | null => {
  for (const h of have) if (h && want.some((w) => eq(h, w))) return h;
  return null;
};

function evalLeaf(leaf: Leaf, c: Candidate, ctx: EvalCtx): Verdict {
  const askerIn = (chat: string) => ctx.askerChats.some((x) => eq(x, chat));
  switch (leaf.field) {
    case "segment": {
      const reasons: string[] = [];
      let ok = false;
      for (const concept of leaf.concepts || []) {
        const chatHit = concept.chats.find((ch) => c.chats.some((x) => eq(x, ch))) || null;
        const modelHit = concept.models.find((m) => c.business_model.some((x) => eq(x, m))) || null;
        const attrHit = concept.attr ? concept.attr(c) : false;
        if (!chatHit && !modelHit && !attrHit) continue;
        ok = true;
        const why: string[] = [];
        if (modelHit && concept.modelReason) why.push(concept.modelReason);
        if (attrHit && concept.attrReason) why.push(concept.attrReason);
        if (chatHit) {
          if (askerIn(chatHit)) why.unshift(`in ${chatHit}`);
          else if (!why.length) why.push(concept.label);      // R10: the chat stays unnamed
        }
        reasons.push(...why);
      }
      return ok ? { ok, reasons } : NO;
    }
    case "chat": {
      const hit = leaf.values.find((ch) => c.chats.some((x) => eq(x, ch))) || null;
      if (!hit) return NO;
      // named only to a member of that chat; the disclosure engine withholds names from anyone else
      return { ok: true, reasons: askerIn(hit) ? [`in ${hit}`] : [] };
    }
    case "event": {
      const hit = leaf.values.find((name) => ctx.eventMembers.get(name)?.has(c.at_member_id)) || null;
      return hit ? { ok: true, reasons: [`attending ${hit}`] } : NO;
    }
    case "city": { const h = pick(leaf.values, [c.city]); return h ? { ok: true, reasons: [h] } : NO; }
    case "state": { const h = pick(leaf.values, [c.state]); return h ? { ok: true, reasons: [h] } : NO; }
    case "country": { const h = pick(leaf.values, [c.country]); return h ? { ok: true, reasons: [h] } : NO; }
    case "band": { const h = pick(leaf.values, [c.rev_band]); return h ? { ok: true, reasons: [h] } : NO; }
    case "niche": { const h = pick(leaf.values, [c.main_niche]); return h ? { ok: true, reasons: [h] } : NO; }
    case "chapter": { const h = pick(leaf.values, c.chapter); return h ? { ok: true, reasons: [h] } : NO; }
    case "category": { const h = pick(leaf.values, c.categories); return h ? { ok: true, reasons: [h] } : NO; }
    case "expertise": {
      const hay = `${c.expertise || ""} ${c.main_niche || ""} ${c.categories.join(" ")}`.toLowerCase();
      const h = leaf.values.find((v) => hay.includes(lower(v))) || null;
      return h ? { ok: true, reasons: [h] } : NO;
    }
    // 🟡 aggregate gates — they decide membership, they never speak (R3).
    case "sku_min": return { ok: Number(c.sku_count || 0) >= Number(leaf.n || 0), reasons: [] };
    case "brands_min": return { ok: Number(c.brands_count || 0) >= Number(leaf.n || 0), reasons: [] };
    case "years_selling_min": {
      const years = c.started_year ? new Date().getFullYear() - c.started_year : -1;
      return { ok: years >= Number(leaf.n || 0), reasons: [] };
    }
    case "age_band": return { ok: !!pick(leaf.values, [c.age_band]), reasons: [] };
    default: return NO;   // a leaf the parser should never have let through
  }
}

/**
 * AND/OR/NOT over the tree. Reasons are the PROOF: every leaf of a satisfied `all`, the satisfied
 * branches of an `any`, nothing from a `not`. The asker is never their own match.
 */
export function evaluate(tree: Tree, c: Candidate, ctx: EvalCtx): Verdict {
  if (c.at_member_id === ctx.askerId) return NO;
  const go = (t: Tree): Verdict => {
    if ("all" in t) {
      const rs = t.all.map(go);
      return rs.every((r) => r.ok) ? { ok: true, reasons: rs.flatMap((r) => r.reasons) } : NO;
    }
    if ("any" in t) {
      const rs = t.any.map(go).filter((r) => r.ok);
      return rs.length ? { ok: true, reasons: rs.flatMap((r) => r.reasons) } : NO;
    }
    if ("not" in t) return { ok: !go(t.not).ok, reasons: [] };
    return evalLeaf(t, c, ctx);
  };
  const v = go(tree);
  return v.ok ? { ok: true, reasons: [...new Set(v.reasons)] } : NO;
}

// ---------------------------------------------------------------------------
// The disclosure engine. Spec §5. Filtering and showing are different rights:
// a 🟡 field may decide WHO is in the set and still never be shown, and naming
// the people a 🟡 filter selected would disclose it anyway — so it doesn't.
// ---------------------------------------------------------------------------

/** #96: a display cap, never a processing cap — `total` always counts everyone. */
export const NAME_CAP = 10;
/** R4: below this, a bucket re-identifies the person the count was meant to protect. */
export const SMALL_BUCKET = 3;

/**
 * #106 (Andy, during #97's prod E2E: "I don't want people to see me as an attendee"). Staff, removed
 * and unknown-status records never reach a member-facing list — and are left out of the count too: a
 * total whose members cannot be shown is a number that lies.
 */
export function isMemberFacing(status: string | null): boolean {
  if (!status) return false;
  const s = status.trim().toLowerCase();
  if (s.startsWith("removed") || s === "staff") return false;
  return s.startsWith("current member") || s === "new member";
}

export function decideDisclosure(x: {
  flags: TreeFlags; askerChats: string[]; askerAttendsAll: boolean; wantNames: boolean;
}): { mayName: boolean; reason: "aggregate_filter" | "not_in_chat" | "not_registered" | null } {
  if (!x.wantNames) return { mayName: false, reason: null };
  if (x.flags.usesAggregate) return { mayName: false, reason: "aggregate_filter" };              // R2
  // R10: listing the members of a chat IS saying who is there — only its members may see that.
  const outside = x.flags.chatLeaves.find((ch) => !x.askerChats.some((a) => a.toLowerCase() === ch.toLowerCase()));
  if (outside) return { mayName: false, reason: "not_in_chat" };
  if (x.flags.eventNames.length && !x.askerAttendsAll) return { mayName: false, reason: "not_registered" }; // R5
  return { mayName: true, reason: null };
}

/** Engagement first, then name. Never surfaced — ordering only, and no equalizer (R9). */
export function rank(people: FoundPerson[]): FoundPerson[] {
  return [...people].sort((a, b) => b.engagement_score - a.engagement_score || a.name.localeCompare(b.name));
}

export function applyCap(people: FoundPerson[], limit: number): {
  people: FoundPerson[]; total: number; shown: number; capped: boolean;
} {
  // FAIL CLOSED: a limit that is not a finite number >= 1 shows NOTHING. The parser supplies the
  // default (NAME_CAP) when the caller omitted it; junk reaching this privacy gate must not widen it.
  const max = Number.isFinite(limit) && limit >= 1 ? Math.min(NAME_CAP, Math.floor(limit)) : 0;
  const ranked = rank(people);
  const shown = ranked.slice(0, max);
  return { people: shown, total: ranked.length, shown: shown.length, capped: ranked.length > shown.length };
}

/** the dimensions groupBy can actually serve — the parser validates group_by against THIS list */
export const GROUP_BY_SERVED = [
  "country", "state", "city", "band", "niche", "business_model", "chat", "chapter", "category", "age_band",
] as const;
export type GroupDimension = (typeof GROUP_BY_SERVED)[number];
export type Bucket = { value: string; count: number | null; note?: string };

export function groupBy(rows: Candidate[], dimension: GroupDimension, suppressSmall: boolean): Bucket[] {
  const pick = (c: Candidate): string[] => {
    switch (dimension) {
      case "country": return c.country ? [c.country] : [];
      case "state": return c.state ? [c.state] : [];
      case "city": return c.city ? [c.city] : [];
      case "band": return c.rev_band ? [c.rev_band] : [];
      case "niche": return c.main_niche ? [c.main_niche] : c.categories.slice(0, 1);
      case "business_model": return c.business_model;
      case "chat": return c.chats;
      case "chapter": return c.chapter;
      case "category": return c.categories;
      case "age_band": return c.age_band ? [c.age_band] : [];
      default: return [];   // a dimension the parser should never have let through — degrade, never throw
    }
  };
  const counts = new Map<string, number>();
  for (const row of rows) for (const v of pick(row)) counts.set(v, (counts.get(v) || 0) + 1);
  return [...counts.entries()]
    .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))
    .map(([value, count]) =>
      suppressSmall && count < SMALL_BUCKET
        ? { value, count: null, note: `fewer than ${SMALL_BUCKET}` }
        : { value, count });
}

// ---------------------------------------------------------------------------
// Request parsing — the closed, class-aware allowlist, applied to every leaf of the tree
// ---------------------------------------------------------------------------

export const RETURN_KINDS = ["people", "count", "breakdown"] as const;   // content | videos | events | partners: phase 2-3
export type ReturnKind = (typeof RETURN_KINDS)[number];
export const MAX_DEPTH = 4;
export const MAX_LEAVES = 20;
/** parsed by the registry, served by a later phase */
export const NOT_SERVED_YET = new Set<string>([...WHAT_KEYS, "speaker_of", "form_answer"]);

export type ParsedRequest =
  | { ok: true; phone: string; where: unknown; ret: ReturnKind; group_by: GroupDimension | null; limit: number }
  | { ok: false; error: string };

function validateNode(node: unknown, depth: number, counter: { leaves: number }): string | null {
  if (depth > MAX_DEPTH) return `tree too deep (max ${MAX_DEPTH})`;
  if (!node || typeof node !== "object" || Array.isArray(node)) return "a node is an object";
  const o = node as Record<string, unknown>;
  const keys = Object.keys(o);
  if (keys.length === 0) return "at least one filter is required";
  // a group node is ONLY its group key — `{all:[…], country:"Spain"}` would silently drop the sibling
  if (keys.some((k) => k === "all" || k === "any" || k === "not") && keys.length !== 1) {
    return "all, any and not take no other keys beside them — nest the condition inside the list";
  }
  if ("all" in o || "any" in o) {
    const list = (o.all ?? o.any) as unknown;
    if (!Array.isArray(list)) return "all and any take a list";
    if (!list.length) return "at least one filter is required";
    for (const child of list) { const e = validateNode(child, depth + 1, counter); if (e) return e; }
    return null;
  }
  if ("not" in o) {
    if (!o.not || typeof o.not !== "object" || Array.isArray(o.not)) return "not takes one node";
    return validateNode(o.not, depth + 1, counter);
  }
  if (keys.length !== 1) return "a leaf holds exactly one field — wrap several in all or any";
  const field = keys[0]; const value = o[field];
  const def = FIELDS[field];
  if (!def) return `unknown filter: ${field}`;
  if (def.cls === "internal") return `filter not allowed: ${field}`;
  if (!def.filterable) return `${field} is a group_by dimension, not a filter — use segment`;
  if (NOT_SERVED_YET.has(field)) return `${field} is not served yet — use content_search or video_search for that`;
  counter.leaves += 1;
  if (counter.leaves > MAX_LEAVES) return `too many conditions (max ${MAX_LEAVES})`;
  if (NUMERIC_FIELDS.has(field)) {
    if (typeof value !== "number" || !Number.isFinite(value)) return `${field} needs a number`;
    return null;
  }
  const vals = asList(value);
  if (!vals.length) return `empty value for ${field}`;
  return null;
}

export function parseRequest(body: unknown): ParsedRequest {
  const b = (body || {}) as Record<string, unknown>;
  const phone = String(b.phone || "").replace(/\D/g, "");
  if (!phone) return { ok: false, error: "phone required" };

  const where = b.where;
  if (!where || typeof where !== "object") return { ok: false, error: "at least one filter is required" };
  const counter = { leaves: 0 };
  const err = validateNode(where, 1, counter);
  if (err) return { ok: false, error: err };

  // The tool schema exposes this as `ret` (the seed is JavaScript, `return` is a keyword); both work.
  const retRaw = String(b.return || b.ret || "people");
  if (["content", "videos", "events", "partners"].includes(retRaw)) return { ok: false, error: `${retRaw} is not served yet` };
  if (!(RETURN_KINDS as readonly string[]).includes(retRaw)) return { ok: false, error: `unknown return: ${retRaw}` };
  const ret = retRaw as ReturnKind;
  const groupRaw = b.group_by ? String(b.group_by) : null;
  if (groupRaw && !GROUP_KEYS.includes(groupRaw)) return { ok: false, error: `unknown group_by: ${groupRaw}` };
  // groupable per the registry but not implemented by groupBy yet (what-group dimensions, phase 2)
  if (groupRaw && !(GROUP_BY_SERVED as readonly string[]).includes(groupRaw)) {
    return { ok: false, error: `group_by ${groupRaw} is not served yet` };
  }
  if (ret === "breakdown" && !groupRaw) return { ok: false, error: "breakdown needs group_by" };

  // missing / junk limit -> the default; applyCap itself fails closed on anything below 1
  const lim = Number(b.limit);
  const limit = Number.isFinite(lim) && lim >= 1 ? Math.floor(lim) : NAME_CAP;
  return { ok: true, phone, where, ret, group_by: (groupRaw as GroupDimension) || null, limit };
}

// DISCLOSURE ENGINE lives here rather than being re-implemented per tool — that divergence is what

export async function POST(req: NextRequest) {
  const secret = process.env.OLIVIA_SCHEDULE_SECRET || process.env.OLIVIA_IOS_SECRET;
  if (!secret) return NextResponse.json({ error: "finder secret not configured" }, { status: 500 });
  const ok = (req.headers.get("authorization") || "") === `Bearer ${secret}` ||
             (req.headers.get("x-olivia-secret") || "") === secret;
  if (!ok) return unauthorized();
  if (!isSupabaseConfigured()) return NextResponse.json({ error: "supabase not configured" }, { status: 500 });

  let raw: unknown;
  try { raw = await req.json(); } catch { return NextResponse.json({ error: "invalid json" }, { status: 400 }); }
  const parsed = parseRequest(raw);
  if (!parsed.ok) return NextResponse.json({ error: parsed.error }, { status: 400 });

  const askerChats = askers[0]?.channels_present || [];   // R10: the chats they may see named

        note: `no event matches "${asked}" — say so plainly and never guess an event` });

    const ev = cands.find((c) => c.name.toLowerCase() === asked.toLowerCase()) ||

    ).catch(() => [] as Array<{ member_at_id: string }>);
    const ids = new Set(regs.map((r) => r.member_at_id));
    events.push(ev); eventMembers.set(ev.name, ids); resolvedName.set(asked, ev.name);
    if (!ids.has(askerId)) askerAttendsAll = false;   // R5: every named event, fail closed

  if (resolvedName.size) {
    tree = mapLeaves(tree, (leaf) => leaf.field === "event"
      ? { ...leaf, values: leaf.values.map((v) => resolvedName.get(v) || v) } : leaf);

    sbRequest<MemberRow[]>("members?select=at_member_id,channels_present,membership_status&at_member_id=not.is.null&limit=3000")
      .catch(() => [] as MemberRow[]),

        .catch(() => [] as ProfRow[]))),
  ]);
  const memberBy = new Map(members.map((m) => [m.at_member_id, m]));
  const profBy = new Map(profChunks.flat().map((p) => [p.at_member_id, p]));
  const candidates: Candidate[] = attrs.map((a) => {
    const m = memberBy.get(a.at_member_id); const p = profBy.get(a.at_member_id);
    return {
      at_member_id: a.at_member_id,
      full_name: a.full_name || p?.full_name || null,
      membership_status: a.membership_status || m?.membership_status || null,
      chats: m?.channels_present || [], business_model: a.business_model || [],
      city: a.city, state: a.state, country: a.country, rev_band: a.rev_band,

      large_sku: a.large_sku, under_30: a.under_30, channel_mix: a.channel_mix || [],
      engagement_score: Number(p?.engagement_score || 0),
    };
  }).filter((c) => isMemberFacing(c.membership_status) && !!c.full_name);   // R8

  // ---- evaluate -----------------------------------------------------------
  const ctx = { askerId, askerChats, eventMembers };
  const matchedRows: Candidate[] = []; const matched: FoundPerson[] = [];
  for (const c of candidates) {
    const v = evaluate(tree, c, ctx);
    if (!v.ok) continue;
    matchedRows.push(c);
    matched.push({ at_member_id: c.at_member_id, name: c.full_name as string, reasons: v.reasons,
                   city: c.city, niche: c.main_niche || c.categories[0] || null, engagement_score: c.engagement_score });
  }

  const disclosure = decideDisclosure({ flags: norm.flags, askerChats, askerAttendsAll, wantNames: ret === "people" });
  const capped = applyCap(matched, limit);
  // R4: small buckets hide when an aggregate leaf filtered the set OR the grouping dimension itself is
  // aggregate-class (age_band) — a one-person "50+" bucket over a small country is a re-identification
  const suppress = norm.flags.usesAggregate || classOf(group_by || "") === "aggregate";
  const breakdown = ret === "breakdown" && group_by ? groupBy(matchedRows, group_by, suppress) : null;

  return NextResponse.json({
    total: capped.total,
    shown: disclosure.mayName ? capped.shown : 0,
    capped: disclosure.mayName ? capped.capped : false,
    where_echo: toEcho(tree),
    unmatched: norm.unmatched,
    disclosure: disclosure.reason === "aggregate_filter" ? "aggregate"
      : disclosure.reason === "not_in_chat" ? "chat"
      : disclosure.reason === "not_registered" ? "event" : "green",
    events,
    people: disclosure.mayName
      ? capped.people.map((p) => ({ name: p.name, reasons: p.reasons, city: p.city, niche: p.niche }))
      : [],
    items: null,
    breakdown,
    note: buildNote({ reason: disclosure.reason, mayName: disclosure.mayName, total: capped.total,
                      shown: capped.shown, unmatched: norm.unmatched, eventNames: events.map((e) => e.name) }),
  });
}

function buildNote(x: {
  reason: string | null; mayName: boolean; total: number; shown: number; unmatched: string[]; eventNames: string[];
}): string {
  const parts: string[] = [];
  if (x.unmatched.length) {
    parts.push(`could not resolve ${x.unmatched.join(", ")} — say which part of the ask you could not honour`);
  }
  if (x.reason === "aggregate_filter") {
    parts.push("this filter uses a group-only attribute, so answer with the number and never name the people — " +
      "do not explain internals, just give the count");
  } else if (x.reason === "not_in_chat") {
    parts.push("this person is not a member of that chat, so give the count only — never the names and never " +
      "what is discussed there — and do not explain internals");
  } else if (x.reason === "not_registered") {
    parts.push(`this person is not registered for ${x.eventNames.join(" / ")}, so answer with the count only, ` +
      "never the names, and do not explain internals");
  } else if (x.mayName && x.total > x.shown) {
    parts.push(`${x.total} people match; ${x.shown} are listed — the count is the truth, never say the listed ` +
      "names are all of them, and offer the rest");
  }
  parts.push("each person carries the reasons they matched, in the order the conditions were asked — say them " +
    "in plain words exactly as given (the tool names a chat only to that chat's members, so never add a chat " +
    "name yourself). Never quote scores or ranks.");
  return parts.join(". ");
}
