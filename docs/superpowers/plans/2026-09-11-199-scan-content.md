// src/lib/millie/team/scan.ts — #199. scan_content: read EVERY row of a member set for a trait no column
// holds. Millie picks the rows with a SELECT (the same way she writes sql_query); this module probes the
// SELECT's shape, counts and prices the set BEFORE reading anything, refuses over the caps, pages the rows
// through digest.team_sql (500 per call, its own cap — no new function, no new grant), classifies 40 rows
// per Haiku call with four calls in flight, stops honestly on the wall or cost cap, and tallies per member.
// Rows are numbered 1..N per batch and mapped back locally; a quote survives only if it is verbatim in the
// row's text. Everything that touches the network is injected (teamSql, classify) so every branch is tested.
import Anthropic from "@anthropic-ai/sdk";
import { config } from "@/lib/config";
import { costUsd, sumUsage, type Usage } from "./pricing";
import { guardSql } from "./sql-guard";

export const SCAN_CAPS = { maxRows: 5000, defaultCostUsd: 2, ceilingCostUsd: 5, batchSize: 40, inFlight: 4, wallMs: 240_000, pageSize: 500, overrunFactor: 1.1 } as const;
export const SCAN_COLUMNS = ["id", "member", "at_member_id", "occurred_at", "text", "url", "source"] as const;
const ROW_CLIP = 1200;
const QUOTE_CLIP = 200;
const RUBRIC_TOKENS = 400;
const OUTPUT_TOKENS_PER_ROW = 20;
const FRAMING_TOKENS_PER_ROW = 15;

export type ScanRow = { id: number; member: string | null; at_member_id: string | null; occurred_at: string | null; text: string; url: string | null; source: string | null };
export type Verdict = "yes" | "no" | "unsure";
export type RowVerdict = { n: number; verdict: Verdict; quote: string };
export type Classify = (question: string, rows: ScanRow[]) => Promise<{ text: string; usage: Usage }>;
export type ScanDeps = { teamSql: (sql: string, maxRows: number) => Promise<{ rows: unknown[]; row_count: number; truncated: boolean }>; classify: Classify; now?: () => number; model?: string };
export type ScanArgs = { sql: string; question: string; maxCostUsd?: number };
export type MemberTally = { member: string; at_member_id: string | null; yes: number; unsure: number; no: number; quotes: Array<{ date: string | null; quote: string; url: string | null; source: string | null }> };
export type ScanResult = { complete: boolean; scanned: number; total_rows: number; cost_usd: number; batches: number; refused?: "too_many_rows" | "over_budget"; error?: string; estimate?: { rows: number; est_cost_usd: number }; members: MemberTally[]; unattributed: { yes: number; unsure: number; no: number } };

const empty = (): ScanResult => ({ complete: false, scanned: 0, total_rows: 0, cost_usd: 0, batches: 0, members: [], unattributed: { yes: 0, unsure: 0, no: 0 } });
const flat = (s: string) => s.replace(/\s+/g, " ").trim();
const round4 = (n: number) => Math.round(n * 10000) / 10000;

export function estimateScan(rows: number, chars: number, model = "claude-haiku-4-5"): number {
  const batches = Math.ceil(rows / SCAN_CAPS.batchSize);
  const input = chars / 4 + rows * FRAMING_TOKENS_PER_ROW + batches * RUBRIC_TOKENS;
  const output = rows * OUTPUT_TOKENS_PER_ROW;
  return round4(costUsd(model, { input_tokens: input, output_tokens: output }));
}

const RUBRIC =
  "You judge messages for ONE question. For each numbered message output exactly one JSON object on its own line: " +
  '{"n": <number>, "verdict": "yes" | "no" | "unsure", "quote": "<verbatim excerpt of at most 200 characters that supports the verdict, or an empty string>"}. ' +
  "yes = the message itself clearly states it. unsure = it hints at it but is not clear. no = anything else. " +
  "Judge only the message's own words — never what you know from elsewhere. Output one line per message, in order, nothing else.";

export function buildBatchPrompt(question: string, rows: ScanRow[]): { system: string; user: string } {
  const lines = rows.map((r, i) => `${i + 1}. ${flat(r.text).slice(0, ROW_CLIP)}`);
  return { system: RUBRIC, user: `QUESTION: ${question}\n\nMESSAGES:\n${lines.join("\n")}` };
}

export function parseVerdicts(text: string, rows: ScanRow[]): RowVerdict[] {
  const out: RowVerdict[] = [];
  const seen = new Set<number>();
  for (const line of text.split("\n")) {
    const s = line.trim();
    if (!s.startsWith("{")) continue;
    let o: { n?: unknown; verdict?: unknown; quote?: unknown };
    try { o = JSON.parse(s); } catch { continue; }
    const n = Number(o.n);
    const verdict = o.verdict;
    if (!Number.isInteger(n) || n < 1 || n > rows.length || seen.has(n)) continue;
    if (verdict !== "yes" && verdict !== "no" && verdict !== "unsure") continue;
    seen.add(n);
    const q = typeof o.quote === "string" ? flat(o.quote).slice(0, QUOTE_CLIP) : "";
    const quote = q && flat(rows[n - 1].text).includes(q) ? q : "";
    out.push({ n, verdict, quote });
  }
  return out;
}

export function tally(rows: ScanRow[], verdicts: Map<number, RowVerdict>): { members: MemberTally[]; unattributed: ScanResult["unattributed"] } {
  const byKey = new Map<string, MemberTally & { _q: Array<{ v: Verdict; date: string | null; quote: string; url: string | null; source: string | null }> }>();
  const unattributed = { yes: 0, unsure: 0, no: 0 };
  for (const r of rows) {
    const v = verdicts.get(r.id);
    if (!v) continue;
    const key = r.at_member_id || r.member;
    if (!key) { unattributed[v.verdict] += 1; continue; }
    let m = byKey.get(key);
    if (!m) { m = { member: r.member || key, at_member_id: r.at_member_id, yes: 0, unsure: 0, no: 0, quotes: [], _q: [] }; byKey.set(key, m); }
    m[v.verdict] += 1;
    if (v.quote && v.verdict !== "no") m._q.push({ v: v.verdict, date: r.occurred_at, quote: v.quote, url: r.url, source: r.source });
  }
  const rank = (v: Verdict) => (v === "yes" ? 0 : 1);
  const members = [...byKey.values()].map((m) => {
    const quotes = m._q.sort((a, b) => rank(a.v) - rank(b.v) || String(b.date ?? "").localeCompare(String(a.date ?? ""))).slice(0, 3).map(({ v: _v, ...q }) => q);
    const { _q: _drop, ...rest } = m;
    return { ...rest, quotes };
  });
  members.sort((a, b) => b.yes - a.yes || b.unsure - a.unsure || a.member.localeCompare(b.member));
  return { members, unattributed };
}

function asRow(o: unknown): ScanRow | null {
  const r = (o && typeof o === "object" ? o : {}) as Record<string, unknown>;
  const id = Number(r.id);
  if (!Number.isFinite(id)) return null;
  return { id, member: r.member == null ? null : String(r.member), at_member_id: r.at_member_id == null ? null : String(r.at_member_id), occurred_at: r.occurred_at == null ? null : String(r.occurred_at), text: r.text == null ? "" : String(r.text), url: r.url == null ? null : String(r.url), source: r.source == null ? null : String(r.source) };
}

/** Run up to `limit` async jobs at a time, in order of submission; stops submitting when `stop()` is true. */
async function pool<T>(jobs: Array<() => Promise<T>>, limit: number, stop: () => boolean): Promise<T[]> {
  const out: T[] = [];
  let i = 0;
  const worker = async () => { while (i < jobs.length && !stop()) { const j = jobs[i++]; out.push(await j()); } };
  await Promise.all(Array.from({ length: Math.min(limit, jobs.length) }, worker));
  return out;
}

export async function runScan(a: ScanArgs, d: ScanDeps): Promise<ScanResult> {
  const now = d.now ?? Date.now;
  const model = d.model ?? "claude-haiku-4-5";
  const t0 = now();
  const res = empty();
  const g = guardSql(a.sql);
  if (!g.ok) return { ...res, error: `refused before the database: ${g.reason}` };
  const inner = g.sql.replace(/;\s*$/, "");
  const cap = Math.min(SCAN_CAPS.ceilingCostUsd, Math.max(0.01, Number(a.maxCostUsd) || SCAN_CAPS.defaultCostUsd));

  // 1. probe the shape — every required column must be present on the first row
  const probe = await d.teamSql(`select * from (${inner}) q limit 1`, 1);
  if (probe.rows.length > 0) {
    const keys = new Set(Object.keys(probe.rows[0] as object));
    const missing = SCAN_COLUMNS.filter((c) => !keys.has(c));
    if (missing.length) return { ...res, error: `the SELECT must return the columns ${SCAN_COLUMNS.join(", ")}; missing: ${missing.join(", ")}` };
  }
  // 2. count and price BEFORE reading anything
  const count = await d.teamSql(`select count(*)::int as n, coalesce(sum(length(text)),0)::bigint as chars from (${inner}) q`, 1);
  const c = (count.rows[0] ?? {}) as { n?: number; chars?: number };
  const total = Number(c.n) || 0;
  const chars = Number(c.chars) || 0;
  const est = estimateScan(total, chars, model);
  res.total_rows = total;
  res.estimate = { rows: total, est_cost_usd: est };
  if (total === 0) return { ...res, complete: true };
  if (total > SCAN_CAPS.maxRows) return { ...res, refused: "too_many_rows" };
  if (est > cap) return { ...res, refused: "over_budget" };

  // 3. page the rows (keyset on id, team_sql's own page cap)
  const rows: ScanRow[] = [];
  let last = 0;
  for (;;) {
    const page = await d.teamSql(`select * from (${inner}) q where q.id > ${last} order by q.id limit ${SCAN_CAPS.pageSize}`, SCAN_CAPS.pageSize);
    const got = page.rows.map(asRow).filter((r): r is ScanRow => r !== null);
    rows.push(...got);
    if (got.length < SCAN_CAPS.pageSize) break;
    last = got[got.length - 1].id;
  }
  // 4. classify in batches, four in flight; empty rows are "no" without a model call
  const verdicts = new Map<number, RowVerdict>();
  for (const r of rows) if (!flat(r.text)) verdicts.set(r.id, { n: 0, verdict: "no", quote: "" });
  const toRead = rows.filter((r) => flat(r.text));
  const batches: ScanRow[][] = [];
  for (let i = 0; i < toRead.length; i += SCAN_CAPS.batchSize) batches.push(toRead.slice(i, i + SCAN_CAPS.batchSize));
  let usage: Usage = { input_tokens: 0, output_tokens: 0 };
  let cost = 0;
  const overrun = () => cost > cap * SCAN_CAPS.overrunFactor || now() - t0 > SCAN_CAPS.wallMs;
  const classifyOnce = async (batch: ScanRow[]) => {
    const r = await d.classify(a.question, batch);
    usage = sumUsage(usage, r.usage);
    cost = costUsd(model, usage);
    return parseVerdicts(r.text, batch);
  };
  const jobs = batches.map((batch) => async () => {
    let v = await classifyOnce(batch);
    if (v.length < batch.length / 2 && !overrun()) v = await classifyOnce(batch); // one retry for an unparseable batch
    for (const x of v) verdicts.set(batch[x.n - 1].id, x);
    return v.length;
  });
  await pool(jobs, SCAN_CAPS.inFlight, overrun);
  res.batches = jobs.length; // attempted batches are what the estimate priced; retries are extra calls
  res.cost_usd = round4(cost);
  res.scanned = verdicts.size;
  res.complete = res.scanned === rows.length;
  const t = tally(rows, verdicts);
  res.members = t.members;
  res.unattributed = t.unattributed;
  return res;
}

export async function classifyWithHaiku(question: string, rows: ScanRow[]): Promise<{ text: string; usage: Usage }> {
  const key = config.centurion.anthropicApiKey;
